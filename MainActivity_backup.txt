@Composable
private fun MindGuardApp() {
    val context = LocalContext.current

    var tabName by rememberSaveable {
        mutableStateOf(Tab.HOME.name)
    }

    val tab =
        Tab.values().firstOrNull { it.name == tabName } ?: Tab.HOME

    var entries by remember {
        mutableStateOf(loadEntries(context))
    }

    var checkInCount by remember {
        mutableStateOf(loadCount(context, entries.size))
    }

    var mood by rememberSaveable { mutableStateOf(3) }
    var stress by rememberSaveable { mutableStateOf(5f) }
    var sleep by rememberSaveable { mutableStateOf(7f) }
    var energy by rememberSaveable { mutableStateOf(6f) }

    var journal by rememberSaveable { mutableStateOf("") }
    var result by remember { mutableStateOf<TextResult?>(null) }

    var familyOptIn by remember {
        mutableStateOf(loadFamilyOptIn(context))
    }

    var showFamilyPreview by rememberSaveable {
        mutableStateOf(false)
    }

    var flState by remember {
        mutableStateOf(FLState())
    }

    val latestSummary =
        entries.lastOrNull()?.summary() ?: "No check-in saved yet"

    Scaffold(
        containerColor = Color.Transparent,
        bottomBar = {
            BottomNavigation(selected = tab) {
                tabName = it.name
            }
        }
    ) { inset ->
        Box(
            modifier = Modifier
                .fillMaxSize()
                .padding(inset)
        ) {
            BackgroundArtwork(Modifier.fillMaxSize())

            key(tab) {
                EnteringScreen {
                    when (tab) {
                        Tab.HOME -> HomeScreen(
                            entries = entries,
                            totalCount = checkInCount,
                            onCheckIn = {
                                tabName = Tab.CHECKIN.name
                            },
                            onAnalyze = {
                                tabName = Tab.ANALYZE.name
                            },
                            onFamily = {
                                tabName = Tab.FAMILY.name
                            },
                            onFL = {
                                tabName = Tab.FL.name
                            }
                        )

                        Tab.CHECKIN -> CheckInScreen(
                            mood = mood,
                            stress = stress,
                            sleep = sleep,
                            energy = energy,
                            onMood = { mood = it },
                            onStress = { stress = it },
                            onSleep = { sleep = it },
                            onEnergy = { energy = it },
                            lastSaved = latestSummary,
                            onSave = {
                                    newMood, newStress,
                                    newSleep, newEnergy ->

                                val entry = CheckIn(
                                    time = System.currentTimeMillis(),
                                    mood = newMood,
                                    stress = newStress,
                                    sleep = newSleep,
                                    energy = newEnergy
                                )

                                entries = (entries + entry).takeLast(30)
                                checkInCount += 1

                                saveEntries(
                                    context,
                                    entries,
                                    checkInCount
                                )
                            }
                        )

                        Tab.ANALYZE -> AnalyzeScreen(
                            text = journal,
                            result = result,
                            onTextChange = {
                                journal = it
                                result = null
                            },
                            onAnalyze = {
                                result = analyzeText(journal)
                            }
                        )

                        Tab.FAMILY -> FamilyScreen(
                            optIn = familyOptIn,
                            summary = latestSummary,
                            showPreview = showFamilyPreview,
                            onOptIn = {
                                familyOptIn = it
                                saveFamilyOptIn(context, it)
                                showFamilyPreview = false
                            },
                            onPreview = {
                                showFamilyPreview = true
                            },
                            onShare = {
                                val shareText =
                                    "Hi, I've been checking in with myself " +
                                            "and could use a little support. " +
                                            "Would you have time to talk with me today?\n\n" +
                                            "My latest reflection: $latestSummary"

                                val intent =
                                    Intent(Intent.ACTION_SEND).apply {
                                        type = "text/plain"
                                        putExtra(Intent.EXTRA_TEXT, shareText)
                                    }

                                context.startActivity(
                                    Intent.createChooser(
                                        intent,
                                        "Share a supportive message"
                                    )
                                )
                            }
                        )

                        Tab.FL -> FederatedScreen(
                            state = flState,
                            onTrain = {
                                flState = FLDemo.runRound(flState)
                            },
                            onReset = {
                                flState = FLState()
                            }
                        )
                    }
                }
            }
        }
    }
}

@Composable
private fun EnteringScreen(
    content: @Composable () -> Unit
) {
    var entered by remember { mutableStateOf(false) }

    val progress by animateFloatAsState(
        targetValue = if (entered) 1f else 0f,
        animationSpec = tween(360),
        label = "screen-enter"
    )

    LaunchedEffect(Unit) {
        entered = true
    }

    Box(
        Modifier.graphicsLayer {
            alpha = progress
            translationY = (1f - progress) * 15f
        }
    ) {
        content()
    }
}

@Composable
private fun BrandMark(size: Dp = 48.dp) {
    val transition = rememberInfiniteTransition(
        label = "brand-pulse"
    )

    val pulse by transition.animateFloat(
        initialValue = 0.94f,
        targetValue = 1.06f,
        animationSpec = infiniteRepeatable(
            animation = tween(1500),
            repeatMode = RepeatMode.Reverse
        ),
        label = "brand-scale"
    )

    Box(
        Modifier.size(size),
        contentAlignment = Alignment.Center
    ) {
        Box(
            Modifier
                .size(size * pulse)
                .background(
                    Color(0x1D68E6C3),
                    CircleShape
                )
        )

        Box(
            Modifier
                .size(size * 0.82f)
                .background(
                    Brush.linearGradient(
                        listOf(MG.mint, MG.blue)
                    ),
                    CircleShape
                ),
            contentAlignment = Alignment.Center
        ) {
            Text(
                "m",
                color = MG.bg,
                fontSize = (size.value * 0.48f).sp,
                fontWeight = FontWeight.Black
            )
        }
    }
}

@Composable
private fun BottomNavigation(
    selected: Tab,
    onSelect: (Tab) -> Unit
) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .navigationBarsPadding()
            .padding(
                start = 12.dp,
                end = 12.dp,
                top = 6.dp,
                bottom = 7.dp
            )
            .clip(RoundedCornerShape(26.dp))
            .background(Color(0xF20D1727))
            .border(
                1.dp,
                MG.border,
                RoundedCornerShape(26.dp)
            )
            .padding(5.dp),
        horizontalArrangement = Arrangement.spacedBy(3.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Tab.values().forEach { item ->
            val active = selected == item

            val scale by animateFloatAsState(
                targetValue = if (active) 1f else 0.94f,
                animationSpec = tween(180),
                label = "nav-scale-${item.name}"
            )

            Column(
                modifier = Modifier
                    .weight(1f)
                    .graphicsLayer {
                        scaleX = scale
                        scaleY = scale
                    }
                    .clip(RoundedCornerShape(19.dp))
                    .background(
                        if (active) Color(0x2468E6C3)
                        else Color.Transparent
                    )
                    .clickable { onSelect(item) }
                    .padding(
                        vertical = 8.dp,
                        horizontal = 1.dp
                    ),
                horizontalAlignment = Alignment.CenterHorizontally,
                verticalArrangement = Arrangement.spacedBy(3.dp)
            ) {
                Text(
                    item.glyph,
                    color = if (active) MG.mint else MG.muted,
                    fontSize = 19.sp,
                    fontWeight = FontWeight.Medium
                )

                Text(
                    item.title,
                    color = if (active) MG.mint else MG.muted,
                    fontSize = 9.sp,
                    fontWeight = if (active) {
                        FontWeight.Bold
                    } else {
                        FontWeight.Medium
                    },
                    maxLines = 1
                )
            }
        }
    }
}

private fun analyzeText(input: String): TextResult {
    val text = input.lowercase(Locale.getDefault()).trim()

    val urgentPhrases = listOf(
        "no reason to live",
        "end it all",
        "can't go on",
        "cant go on",
        "want to disappear"
    )

    val terms = listOf(
        "overwhelmed", "anxious", "anxiety", "panic",
        "stressed", "stressful", "exhausted", "lonely",
        "alone", "can't sleep", "cant sleep", "worried",
        "pressure", "burnt out", "burned out", "tired",
        "frustrated", "sad", "scared"
    ).filter { text.contains(it) }.distinct()

    val urgent = urgentPhrases.any { text.contains(it) }

    return when {
        urgent -> TextResult(
            "Please reach for support",
            "Support may be important",
            "Some wording may be associated with significant distress. " +
                    "This keyword tool cannot understand your whole situation or assess your safety.",
            "Consider contacting someone you trust now. If you may be in immediate danger, " +
                    "contact local emergency services or a crisis support service in your area.",
            terms.size,
            true
        )

        terms.size >= 3 -> TextResult(
            "A lot may be on your mind",
            "Several reflection signals",
            "This entry contains ${terms.size} terms from a small stress-related list: " +
                    "${terms.take(4).joinToString(", ")}. Context matters; this is not a diagnosis.",
            "Try identifying one thing you can control today, take a short pause, " +
                    "or speak with someone supportive.",
            terms.size,
            false
        )

        terms.isNotEmpty() -> TextResult(
            "Pause and check in with yourself",
            "A few reflection signals",
            "This entry contains: ${terms.joinToString(", ")}. A phrase alone cannot capture your whole experience.",
            "Consider what might help in this moment: rest, food or water, a break, movement, or a conversation.",
            terms.size,
            false
        )

        else -> TextResult(
            "Your entry is more than a score",
            "No listed terms found",
            "This demo did not find terms from its small stress-related list. It can miss context and emotions; this does not mean everything is fine.",
            "Use your own judgement about how you feel. You do not need a keyword result to ask for support.",
            0,
            false
        )
    }
}

private object FLDemo {
    private val clientA = listOf(
        Example(doubleArrayOf(1.0, 0.10, 0.10, 0.10), 0.0),
        Example(doubleArrayOf(1.0, 0.20, 0.10, 0.20), 0.0),
        Example(doubleArrayOf(1.0, 0.25, 0.25, 0.10), 0.0),
        Example(doubleArrayOf(1.0, 0.35, 0.10, 0.25), 0.0),
        Example(doubleArrayOf(1.0, 0.65, 0.45, 0.60), 1.0),
        Example(doubleArrayOf(1.0, 0.80, 0.50, 0.70), 1.0),
        Example(doubleArrayOf(1.0, 0.75, 0.80, 0.55), 1.0),
        Example(doubleArrayOf(1.0, 0.60, 0.30, 0.80), 1.0)
    )

    private val clientB = listOf(
        Example(doubleArrayOf(1.0, 0.10, 0.05, 0.10), 0.0),
        Example(doubleArrayOf(1.0, 0.15, 0.25, 0.10), 0.0),
        Example(doubleArrayOf(1.0, 0.30, 0.20, 0.30), 0.0),
        Example(doubleArrayOf(1.0, 0.40, 0.15, 0.20), 0.0),
        Example(doubleArrayOf(1.0, 0.60, 0.65, 0.55), 1.0),
        Example(doubleArrayOf(1.0, 0.70, 0.70, 0.65), 1.0),
        Example(doubleArrayOf(1.0, 0.85, 0.45, 0.75), 1.0),
        Example(doubleArrayOf(1.0, 0.65, 0.35, 0.90), 1.0)
    )

    private val validation = listOf(
        Example(doubleArrayOf(1.0, 0.12, 0.10, 0.08), 0.0),
        Example(doubleArrayOf(1.0, 0.25, 0.22, 0.20), 0.0),
        Example(doubleArrayOf(1.0, 0.38, 0.18, 0.28), 0.0),
        Example(doubleArrayOf(1.0, 0.64, 0.52, 0.65), 1.0),
        Example(doubleArrayOf(1.0, 0.74, 0.62, 0.65), 1.0),
        Example(doubleArrayOf(1.0, 0.78, 0.40, 0.82), 1.0)
    )

    fun runRound(previous: FLState): FLState {
        val a = train(previous.weights, clientA)
        val b = train(previous.weights, clientB)

        val total = clientA.size + clientB.size

        val global = previous.weights.indices.map { i ->
            (
                    a[i] * clientA.size +
                            b[i] * clientB.size
                    ) / total.toDouble()
        }

        val (accuracy, loss) = evaluate(global, validation)

        return FLState(
            rounds = previous.rounds + 1,
            weights = global,
            clientA = a,
            clientB = b,
            accuracy = accuracy,
            loss = loss
        )
    }

    private fun train(
        initial: List<Double>,
        data: List<Example>
    ): List<Double> {
        val w = initial.toDoubleArray()
        val rate = 0.32

        repeat(32) {
            val gradient = DoubleArray(w.size)

            data.forEach { example ->
                val error =
                    sigmoid(dot(w, example.features)) - example.label

                w.indices.forEach { i ->
                    gradient[i] += error * example.features[i]
                }
            }

            w.indices.forEach { i ->
                val reg = if (i == 0) 0.0 else 0.008 * w[i]
                w[i] -= rate * (
                        gradient[i] / data.size.toDouble() + reg
                        )
            }
        }

        return w.toList()
    }

    private fun evaluate(
        weights: List<Double>,
        data: List<Example>
    ): Pair<Double, Double> {
        var correct = 0
        var loss = 0.0

        data.forEach { example ->
            val p = sigmoid(
                dot(weights.toDoubleArray(), example.features)
            ).coerceIn(1e-7, 1.0 - 1e-7)

            if ((if (p >= 0.5) 1.0 else 0.0) == example.label) {
                correct++
            }

            loss += -(
                    example.label * ln(p) +
                            (1.0 - example.label) * ln(1.0 - p)
                    )
        }

        return (correct.toDouble() / data.size.toDouble()) to
                (loss / data.size.toDouble())
    }

    private fun dot(
        weights: DoubleArray,
        features: DoubleArray
    ): Double =
        weights.indices.sumOf { weights[it] * features[it] }

    private fun sigmoid(value: Double): Double =
        1.0 / (1.0 + exp(-value.coerceIn(-30.0, 30.0)))
}

private fun loadEntries(context: Context): List<CheckIn> {
    val raw = context
        .getSharedPreferences("mindguard_demo", Context.MODE_PRIVATE)
        .getString("entries", "")
        .orEmpty()

    return raw.split(';').mapNotNull { row ->
        val p = row.split(',')

        if (p.size != 5) {
            null
        } else {
            runCatching {
                CheckIn(
                    p[0].toLong(),
                    p[1].toInt(),
                    p[2].toInt(),
                    p[3].toInt(),
                    p[4].toInt()
                )
            }.getOrNull()
        }
    }.takeLast(30)
}

private fun loadCount(
    context: Context,
    fallback: Int
): Int =
    context
        .getSharedPreferences("mindguard_demo", Context.MODE_PRIVATE)
        .getInt("count", fallback)
        .coerceAtLeast(fallback)

private fun saveEntries(
    context: Context,
    entries: List<CheckIn>,
    count: Int
) {
    val serialized = entries.joinToString(";") {
        "${it.time},${it.mood},${it.stress},${it.sleep},${it.energy}"
    }

    context
        .getSharedPreferences("mindguard_demo", Context.MODE_PRIVATE)
        .edit()
        .putString("entries", serialized)
        .putInt("count", count)
        .apply()
}
